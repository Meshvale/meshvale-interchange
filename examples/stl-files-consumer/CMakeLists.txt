cmake_minimum_required(VERSION 3.24)
project(MeshvaleStlFilesInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleInterchange CONFIG REQUIRED)
if(NOT TARGET meshvale::stl_files)
  message(FATAL_ERROR "The installed Interchange package lacks optional meshvale::stl_files")
endif()
add_executable(meshvale_stl_files_consumer ../../tests/stl_files_consumer.cpp)
target_include_directories(meshvale_stl_files_consumer PRIVATE ../../tests)
target_compile_features(meshvale_stl_files_consumer PRIVATE cxx_std_20)
target_link_libraries(meshvale_stl_files_consumer PRIVATE meshvale::stl_files)
if(MSVC)
  target_compile_options(meshvale_stl_files_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(meshvale_stl_files_consumer PRIVATE
    -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MaterializationTestProfile.cmake")
include(CTest)
if(MESHVALE_INTERCHANGE_MATERIALIZATION_TEST_PROFILE STREQUAL "unsupported")
  add_test(NAME installed_stl_files_consumer COMMAND meshvale_stl_files_consumer --unsupported-profile)
else()
  add_test(NAME installed_stl_files_consumer COMMAND meshvale_stl_files_consumer)
endif()
