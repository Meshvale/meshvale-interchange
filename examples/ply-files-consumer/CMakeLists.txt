cmake_minimum_required(VERSION 3.24)
project(MeshvalePlyFilesInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleInterchange CONFIG REQUIRED)
if(NOT TARGET meshvale::ply_files)
  message(FATAL_ERROR "Install MeshvaleInterchange with PLY file support enabled")
endif()
add_executable(meshvale_ply_files_consumer main.cpp)
target_compile_features(meshvale_ply_files_consumer PRIVATE cxx_std_20)
target_link_libraries(meshvale_ply_files_consumer PRIVATE meshvale::ply_files)
if(MSVC)
  target_compile_options(meshvale_ply_files_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(meshvale_ply_files_consumer PRIVATE
    -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
include(../../cmake/MaterializationTestProfile.cmake)
include(CTest)
if(MESHVALE_INTERCHANGE_MATERIALIZATION_TEST_PROFILE STREQUAL "qualified")
  add_test(NAME installed_ply_files_consumer COMMAND meshvale_ply_files_consumer)
else()
  add_test(NAME installed_ply_files_consumer
    COMMAND meshvale_ply_files_consumer --unsupported-profile)
endif()
