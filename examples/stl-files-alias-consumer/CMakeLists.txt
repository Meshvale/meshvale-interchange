cmake_minimum_required(VERSION 3.24)
project(MeshvaleStlFilesAliasInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32)
  message(FATAL_ERROR "The explicit STL path-alias consumer requires Windows")
endif()
find_package(MeshvaleInterchange CONFIG REQUIRED)
if(NOT TARGET meshvale::stl_files)
  message(FATAL_ERROR "The installed Interchange package lacks optional meshvale::stl_files")
endif()
add_executable(meshvale_stl_files_alias_consumer ../../tests/stl_files_aliases.cpp)
target_include_directories(meshvale_stl_files_alias_consumer PRIVATE ../../tests)
target_compile_features(meshvale_stl_files_alias_consumer PRIVATE cxx_std_20)
target_link_libraries(meshvale_stl_files_alias_consumer PRIVATE meshvale::stl_files)
if(MSVC)
  target_compile_options(meshvale_stl_files_alias_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(meshvale_stl_files_alias_consumer PRIVATE
    -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
