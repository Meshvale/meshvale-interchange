cmake_minimum_required(VERSION 3.24)
project(MeshvaleOffInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleInterchange CONFIG REQUIRED)
if(NOT TARGET meshvale::off)
  message(FATAL_ERROR "The installed MeshvaleInterchange package lacks meshvale::off")
endif()
add_executable(meshvale_off_consumer main.cpp)
target_compile_features(meshvale_off_consumer PRIVATE cxx_std_20)
target_link_libraries(meshvale_off_consumer PRIVATE meshvale::off)
if(MSVC)
  target_compile_options(meshvale_off_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(meshvale_off_consumer PRIVATE
    -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MaterializationTestProfile.cmake")
include(CTest)
if(MESHVALE_INTERCHANGE_MATERIALIZATION_TEST_PROFILE STREQUAL "unsupported")
  add_test(NAME installed_off_consumer COMMAND meshvale_off_consumer --unsupported-profile)
else()
  add_test(NAME installed_off_consumer COMMAND meshvale_off_consumer)
endif()
