if(MESHVALE_INTERCHANGE_GLTF_FILES)
  if(NOT TARGET meshvale::gltf_files)
    message(FATAL_ERROR "Requested installed glTF files target is unavailable")
  endif()
  add_executable(meshvale_gltf_files_link_consumer gltf_files_link.cpp)
  add_executable(meshvale_gltf_files_consumer gltf_files.cpp)
  foreach(consumer IN ITEMS meshvale_gltf_files_link_consumer meshvale_gltf_files_consumer)
    target_link_libraries(${consumer} PRIVATE meshvale::gltf_files)
    if(MSVC)
      target_compile_options(${consumer} PRIVATE /W4 /WX)
    else()
      target_compile_options(${consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endforeach()
  add_test(NAME installed_gltf_files_link COMMAND meshvale_gltf_files_link_consumer)
  add_test(NAME installed_gltf_files_consumer COMMAND meshvale_gltf_files_consumer)
endif()
