cmake_minimum_required(VERSION 3.24)
project(MeshvaleFbxConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleInterchangeFbx 0.0.0 EXACT CONFIG REQUIRED)
add_executable(fbx_consumer main.cpp)
target_link_libraries(fbx_consumer PRIVATE meshvale::interchange_fbx)
target_compile_options(fbx_consumer PRIVATE /W4 /WX)
set_target_properties(fbx_consumer PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
enable_testing()
set(FBX_INPUT "" CACHE FILEPATH "Caller-supplied supported FBX file")
if(FBX_INPUT)
  add_test(NAME fbx_installed_consumer COMMAND fbx_consumer "${FBX_INPUT}")
endif()
