cmake_minimum_required(VERSION 3.24)
project(MeshvaleStlInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleInterchange CONFIG REQUIRED)
if(NOT TARGET meshvale::stl)
  message(FATAL_ERROR "The installed MeshvaleInterchange package lacks meshvale::stl")
endif()
add_executable(meshvale_stl_consumer main.cpp)
target_compile_features(meshvale_stl_consumer PRIVATE cxx_std_20)
target_link_libraries(meshvale_stl_consumer PRIVATE meshvale::stl)
if(MSVC)
  target_compile_options(meshvale_stl_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(meshvale_stl_consumer PRIVATE
    -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MaterializationTestProfile.cmake")
include(CTest)
if(MESHVALE_INTERCHANGE_MATERIALIZATION_TEST_PROFILE STREQUAL "unsupported")
  add_test(NAME installed_stl_consumer COMMAND meshvale_stl_consumer --unsupported-profile)
else()
  add_test(NAME installed_stl_consumer COMMAND meshvale_stl_consumer)
endif()
