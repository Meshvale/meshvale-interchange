cmake_minimum_required(VERSION 3.24)
project(MeshvalePlyInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleInterchange CONFIG REQUIRED)
find_package(MeshvaleRepair CONFIG REQUIRED)
add_executable(meshvale_ply_repair_consumer main.cpp)
target_compile_features(meshvale_ply_repair_consumer PRIVATE cxx_std_20)
target_link_libraries(meshvale_ply_repair_consumer PRIVATE meshvale::ply meshvale::materialized_duplicates)
if(MSVC)
  target_compile_options(meshvale_ply_repair_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(meshvale_ply_repair_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/MaterializationTestProfile.cmake")
include(CTest)
if(MESHVALE_INTERCHANGE_MATERIALIZATION_TEST_PROFILE STREQUAL "unsupported")
  add_test(NAME installed_ply_repair_consumer COMMAND meshvale_ply_repair_consumer --unsupported-profile)
else()
  add_test(NAME installed_ply_repair_consumer COMMAND meshvale_ply_repair_consumer)
endif()
