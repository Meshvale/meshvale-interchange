cmake_minimum_required(VERSION 3.24)
project(MeshvaleOffRepairInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleInterchange CONFIG REQUIRED)
find_package(MeshvaleRepair CONFIG REQUIRED)
if(NOT TARGET meshvale::off OR NOT TARGET meshvale::materialized_duplicates)
  message(FATAL_ERROR "Install the native OFF and charged duplicate Repair targets")
endif()
add_executable(meshvale_off_repair_consumer main.cpp)
target_compile_features(meshvale_off_repair_consumer PRIVATE cxx_std_20)
target_link_libraries(meshvale_off_repair_consumer PRIVATE
  meshvale::off meshvale::materialized_duplicates)
if(MSVC)
  target_compile_options(meshvale_off_repair_consumer PRIVATE /W4 /WX /fp:strict)
else()
  target_compile_options(meshvale_off_repair_consumer PRIVATE
    -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
include(../../cmake/MaterializationTestProfile.cmake)
include(CTest)
if(MESHVALE_INTERCHANGE_MATERIALIZATION_TEST_PROFILE STREQUAL "qualified")
  add_test(NAME installed_off_repair_consumer COMMAND meshvale_off_repair_consumer)
else()
  add_test(NAME installed_off_repair_consumer COMMAND meshvale_off_repair_consumer --unsupported-profile)
endif()
