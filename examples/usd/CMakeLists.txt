cmake_minimum_required(VERSION 3.24)
project(MeshvaleUsdConsumer LANGUAGES CXX)
find_package(MeshvaleInterchangeUsd 0.0.0 EXACT CONFIG REQUIRED)
add_executable(meshvale_usd_consumer main.cpp)
set_target_properties(meshvale_usd_consumer PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(meshvale_usd_consumer PRIVATE meshvale::interchange_usd)
if(MSVC)
  target_compile_options(meshvale_usd_consumer PRIVATE /W4 /WX)
else()
  target_compile_options(meshvale_usd_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(DEFINED USD_MESH_FILE)
  include(CTest)
  add_test(NAME installed_usd_mesh COMMAND meshvale_usd_consumer "${USD_MESH_FILE}" /Mesh)
endif()
